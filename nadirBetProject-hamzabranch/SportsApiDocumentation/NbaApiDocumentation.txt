NbaApiDocumentation

1. Base Configuration
{
  "base_url": "https://v2.nba.api-sports.io/",
  "method": "GET",
  "auth_header": "x-apisports-key",
  "format": "JSON"
}
💻 cURL Example
curl -X GET "https://v2.nba.api-sports.io/status" \
  -H "x-apisports-key: YOUR_API_KEY"

🏀 4. Teams
[
  {
    "endpoint": "/teams",
    "description": "NBA teams information",
    "params": ["id", "name", "league", "season", "country", "search"]
  },
  {
    "endpoint": "/teams/statistics",
    "description": "Team statistics (offense, defense, wins, losses)",
    "params": ["team", "season", "league"]
  }
]
👥 5. Players
[
  {
    "endpoint": "/players",
    "description": "NBA players list and profiles",
    "params": ["id", "team", "season", "search", "page"]
  },
  {
    "endpoint": "/players/statistics",
    "description": "Player season statistics",
    "params": ["id", "team", "season", "league"]
  }
]
🏟️ 6. Games (Fixtures)
[
  {
    "endpoint": "/games",
    "description": "NBA games schedule and results",
    "params": ["id", "league", "season", "team", "date", "live", "timezone"]
  },
  {
    "endpoint": "/games/statistics",
    "description": "Game statistics (team comparison, stats breakdown)",
    "params": ["id", "team"]
  },
  {
    "endpoint": "/games/events",
    "description": "Game events (scoring timeline, fouls, etc.)",
    "params": ["id"]
  },
  {
    "endpoint": "/games/lineups",
    "description": "Game lineups (starting players, rotations)",
    "params": ["id"]
  },
  {
    "endpoint": "/games/h2h",
    "description": "Head-to-head match history",
    "params": ["h2h", "season", "league"]
  }
]
