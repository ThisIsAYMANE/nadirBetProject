RugbyApiDocumentation

1. Base Configuration
{
  "base_url": "https://v1.rugby.api-sports.io/",
  "method": "GET",
  "auth_header": "x-apisports-key",
  "format": "JSON"
}
💻 cURL Example
curl -X GET "https://v1.rugby.api-sports.io/status" \
  -H "x-apisports-key: YOUR_API_KEY"

🏟️ 4. Teams
[
  {
    "endpoint": "/teams",
    "description": "Teams information",
    "params": ["id", "name", "league", "season", "country", "search"]
  },
  {
    "endpoint": "/teams/statistics",
    "description": "Team statistics (wins, losses, performance)",
    "params": ["team", "league", "season"]
  }
]
👥 5. Players
[
  {
    "endpoint": "/players",
    "description": "Players list and profiles",
    "params": ["id", "team", "league", "season", "search", "page"]
  },
  {
    "endpoint": "/players/statistics",
    "description": "Player performance stats",
    "params": ["id", "team", "season", "league"]
  }
]
🏉 6. Games (Fixtures)
[
  {
    "endpoint": "/games",
    "description": "Matches schedule and results",
    "params": ["id", "league", "season", "team", "date", "live", "timezone"]
  },
  {
    "endpoint": "/games/statistics",
    "description": "Match statistics",
    "params": ["id"]
  },
  {
    "endpoint": "/games/events",
    "description": "Match events (tries, penalties, cards)",
    "params": ["id"]
  },
  {
    "endpoint": "/games/teams",
    "description": "Teams in a match",
    "params": ["id"]
  },
  {
    "endpoint": "/games/players",
    "description": "Player stats per match",
    "params": ["id"]
  },
  {
    "endpoint": "/games/h2h",
    "description": "Head-to-head matches",
    "params": ["h2h", "league", "season"]
  }
]

🧾 11. Standard Response Format
{
  "get": "endpoint_name",
  "parameters": {},
  "errors": [],
  "results": 0,
  "paging": {
    "current": 1,
    "total": 1
  },
  "response": []
}
